cmake_minimum_required(VERSION 3.16)
project(well_e_closed_loop)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
add_executable(closed_loop nodes.cpp)
target_compile_features(closed_loop PUBLIC cxx_std_17)
target_compile_options(closed_loop PRIVATE -O3 -Wall)
target_link_libraries(closed_loop Eigen3::Eigen OpenMP::OpenMP_CXX)
ament_target_dependencies(closed_loop rclcpp rclcpp_action std_msgs nav_msgs nav2_msgs geometry_msgs)
install(TARGETS closed_loop DESTINATION lib/${PROJECT_NAME})
add_executable(score_benchmark score_benchmark.cpp)
target_compile_features(score_benchmark PUBLIC cxx_std_17)
target_compile_options(score_benchmark PRIVATE -O3 -Wall)
target_link_libraries(score_benchmark Eigen3::Eigen OpenMP::OpenMP_CXX)
install(TARGETS score_benchmark DESTINATION lib/${PROJECT_NAME})
ament_package()
